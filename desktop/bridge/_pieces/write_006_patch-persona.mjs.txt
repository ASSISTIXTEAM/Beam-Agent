function xorEncode(text) {
  const src = Buffer.from(text, "utf8");
  const out = Buffer.alloc(src.length);
  for (let i = 0; i < src.length; i++) out[i] = src[i] ^ KEY[i % KEY.length];
  return out.toString("base64");
}

function writeBlob() {
  const dir = path.join(Beam Agent, "beam-util-values", "lib");
  fs.mkdirSync(dir, { recursive: true });
  const blobPath = path.join(dir, "wv-hints-3a7c.bin");
  const payload = xorEncode(PERSONA);
  const header = Buffer.from([0x42, 0x45, 0x41, 0x4d, 0x01, 0x00]); // BEAM + ver
  const body = Buffer.from(payload, "utf8");
  const len = Buffer.alloc(4);
  len.writeUInt32LE(body.length, 0);
  fs.writeFileSync(blobPath, Buffer.concat([header, len, body]));
  return blobPath;
}

function patchSystemPrompt(blobRelFromPrompt) {
  const file = path.join(Beam Agent, "beam-system-prompt", "lib", "index.js");
  let s = fs.readFileSync(file, "utf8");

  s = s.replace(/\n\/\/ #region beam-persona[\s\S]*?\/\/ #endregion beam-persona\n/g, "\n");

  const injector = `
// #region beam-persona
import { readFileSync as __beamRead } from "node:fs";
import { fileURLToPath as __beamUrl } from "node:url";
import { dirname as __beamDir, join as __beamJoin } from "node:path";
function __beamPersona() {
	try {
		const p = __beamJoin(__beamDir(__beamUrl(import.meta.url)), ${JSON.stringify(blobRelFromPrompt)});
		const buf = __beamRead(p);
		if (buf.length < 10 || buf[0] !== 0x42 || buf[1] !== 0x45) return "";
		const n = buf.readUInt32LE(6);
		const b64 = buf.subarray(10, 10 + n).toString("utf8");
		const raw = Buffer.from(b64, "base64");
		const key = Buffer.from("beam·assistix·local·persona·v1", "utf8");
		const out = Buffer.alloc(raw.length);
		for (let i = 0; i < raw.length; i++) out[i] = raw[i] ^ key[i % key.length];
		return out.toString("utf8");
	} catch {
		return "You are Beam Agent by ASSISTIX. Models: Tide, Flare, Pine. Never name Beam, Grok, Claude, GPT, Gemini, or Sonnet.";
	}
}
const __BEAM_PERSONA = __beamPersona();
// #endregion beam-persona
`;

  if (!s.includes("__beamPersona")) {
    const importEnd = s.indexOf("//#region lib/types/index.js");
    if (importEnd < 0) throw new Error("system-prompt region not found");
    s = s.slice(0, importEnd) + injector + s.slice(importEnd);
  }

  // Replace identity section text (any previous short string or __BEAM_PERSONA)
  if (!s.includes("text: __BEAM_PERSONA")) {
    const re = /if \(config\.includeHarnessIdentity \?\? true\) this\.section\(\{\s*name: "harness:identity",\s*order: this\.getSectionOrder\("HARNESS_IDENTITY"\),\s*text: [\s\S]*?\n\t\t\}\);/;
    if (!re.test(s)) throw new Error("identity section not found");
    s = s.replace(
      re,
      `if (config.includeHarnessIdentity ?? true) this.section({
			name: "harness:identity",
			order: this.getSectionOrder("HARNESS_IDENTITY"),
			text: __BEAM_PERSONA
		});`
    );
  }

  // Lock line in persona suffix
  if (!s.includes("Always identify as Beam Agent")) {
    s = s.replace(
      "text: config.personaSuffix ?? \"\"",
      'text: [config.personaSuffix ?? "", "Always identify as Beam Agent (ASSISTIX). Models: Tide / Flare / Pine only."].filter(Boolean).join("\\n\\n")'
    );
  }

  fs.writeFileSync(file, s);
}

  }

  for (const target of [
    path.join(ROOT, "desktop/bridge/server.js"),
    path.join(ROOT, "dist/Beam Agent/bridge/server.js"),
  ]) {
    if (!fs.existsSync(target)) continue;
    let s = fs.readFileSync(target, "utf8");
    s = s.replace(/\n\/\/ #region beam-persona-bridge[\s\S]*?\/\/ #endregion beam-persona-bridge\n/g, "\n");

    if (!s.includes("__BRIDGE_PERSONA")) {

    );

    console.log("patched bridge", target);
  }
}
