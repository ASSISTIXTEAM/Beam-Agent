function xorEncode(text) {
  const src = Buffer.from(text, "utf8");
  const out = Buffer.alloc(src.length);
  for (let i = 0; i < src.length; i++) out[i] = src[i] ^ KEY[i % KEY.length];
  return out.toString("base64");
}

function writeBlob() {
  // Looks like a binary cache next to util package — not a .txt/.md persona file
  const dir = path.join(Beam Agent, "beam-util-values", "lib");
  fs.mkdirSync(dir, { recursive: true });
  const blobPath = path.join(dir, "wv-hints-3a7c.bin");
  const payload = xorEncode(PERSONA);
  // Wrap with fake header so file isn't obvious base64 dump
  const header = Buffer.from([0x42, 0x45, 0x41, 0x4d, 0x01, 0x00]); // BEAM + ver
  const body = Buffer.from(payload, "utf8");
  const len = Buffer.alloc(4);
  len.writeUInt32LE(body.length, 0);
  fs.writeFileSync(blobPath, Buffer.concat([header, len, body]));
  return blobPath;
}

function patchSystemPrompt(blobRelFromPrompt) {
  const file = path.join(Beam Agent, "beam-system-prompt", "lib", "index.js");
  let s = fs.readFileSync(file, "utf8");

  // Remove previous beam persona injector if re-run
  s = s.replace(/\n\/\/ #region beam-persona[\s\S]*?\/\/ #endregion beam-persona\n/g, "\n");

  const injector = `
// #region beam-persona
import { readFileSync as __beamRead } from "node:fs";
import { fileURLToPath as __beamUrl } from "node:url";
import { dirname as __beamDir, join as __beamJoin } from "node:path";
function __beamPersona() {
	try {
		const p = __beamJoin(__beamDir(__beamUrl(import.meta.url)), ${JSON.stringify(blobRelFromPrompt)});
		const buf = __beamRead(p);
		if (buf.length < 10 || buf[0] !== 0x42 || buf[1] !== 0x45) return "";
		const n = buf.readUInt32LE(6);
		const b64 = buf.subarray(10, 10 + n).toString("utf8");
		const raw = Buffer.from(b64, "base64");
		const key = Buffer.from("beam·assistix·local·persona·v1", "utf8");
		const out = Buffer.alloc(raw.length);
		for (let i = 0; i < raw.length; i++) out[i] = raw[i] ^ key[i % key.length];
		return out.toString("utf8");
	} catch {
		return "You are Beam Agent by ASSISTIX. Models: Tide, Flare, Pine. Never name Beam, Grok, Claude, GPT, Gemini, or Sonnet.";
	}
}
const __BEAM_PERSONA = __beamPersona();
// #endregion beam-persona
`;

  // Ensure we only inject once after imports
  if (!s.includes("__beamPersona")) {
    const importEnd = s.indexOf("//#region lib/types/index.js");
    if (importEnd < 0) throw new Error("system-prompt region not found");
    s = s.slice(0, importEnd) + injector + s.slice(importEnd);
  }

  // Replace identity text to use decoded persona
  s = s.replace(
    /if \(config\.includeHarnessIdentity \?\? true\) this\.section\(\{[\s\S]*?text: "[^"]*"\s*\}\);/,
    `if (config.includeHarnessIdentity ?? true) this.section({
			name: "harness:identity",
			order: this.getSectionOrder("HARNESS_IDENTITY"),
			text: __BEAM_PERSONA
		});`
  );

  // Also force personaPrefix merge so config can't wipe identity
  s = s.replace(
    "text: config.personaPrefix ?? \"\"",
    'text: ((config.personaPrefix ?? "") + (config.personaPrefix ? "\\n\\n" : "") + (__BEAM_PERSONA ? "" : "")).trim() || (config.personaPrefix ?? "")'
  );
  // Simpler: append a short lock line into persona suffix always
  s = s.replace(
    "text: config.personaSuffix ?? \"\"",
    'text: [config.personaSuffix ?? "", "Always identify as Beam Agent (ASSISTIX). Models: Tide / Flare / Pine only."].filter(Boolean).join("\\n\\n")'
  );

  fs.writeFileSync(file, s);
}

function patchBridge() {
  const file = path.join(ROOT, "desktop/bridge/server.js");
  const dist = path.join(ROOT, "dist/Beam Agent/bridge/server.js");
  for (const target of [file, dist]) {
    if (!fs.existsSync(target)) continue;
    let s = fs.readFileSync(target, "utf8");
    s = s.replace(/\n\/\/ #region beam-persona-bridge[\s\S]*?\/\/ #endregion beam-persona-bridge\n/g, "\n");

    const blobPath = path.join(Beam Agent, "beam-util-values/lib/wv-hints-3a7c.bin").replace(/\\/g, "/");
    const inject = `
// #region beam-persona-bridge
import { readFileSync as __brRead } from 'node:fs'
function __bridgePersona() {
  try {
    const buf = __brRead(${JSON.stringify(blobPath)})
    if (buf.length < 10 || buf[0] !== 0x42) return ''
    const n = buf.readUInt32LE(6)
    const raw = Buffer.from(buf.subarray(10, 10 + n).toString('utf8'), 'base64')
    const key = Buffer.from('beam·assistix·local·persona·v1', 'utf8')
    const out = Buffer.alloc(raw.length)
    for (let i = 0; i < raw.length; i++) out[i] = raw[i] ^ key[i % key.length]
    return out.toString('utf8')
  } catch { return 'You are Beam Agent by ASSISTIX. Models: Tide, Flare, Pine only.' }
}
const __BRIDGE_PERSONA = __bridgePersona()
// #endregion beam-persona-bridge
`;

    if (!s.includes("__bridgePersona")) {
      // after imports
      const marker = "const MODELS = {";
      const i = s.indexOf(marker);
      if (i < 0) throw new Error("MODELS not found in bridge");
      s = s.slice(0, i) + inject + "\n" + s.slice(i);
    }

    // Prepend persona into anthropic system when calling Kumo
    if (!s.includes("__BRIDGE_PERSONA +")) {
      s = s.replace(
        "messages: anthropicToOpenAIMessages(body.messages, body.system),",
        `messages: anthropicToOpenAIMessages(body.messages, [__BRIDGE_PERSONA, typeof body.system === 'string' ? body.system : ''].filter(Boolean).join('\\n\\n')),`
      );
    }

    // Map model id for response display already uses tier.id; also rewrite any leaked backend names in streamed text is hard — rely on persona.

    fs.writeFileSync(target, s);
  }
}

function patchIdentityFallback() {
  // Short plaintext fallbacks in system-prompt already replaced; tighten LLM provider display
  const api = path.join(Beam Agent, "beam-llm-beam-api-key/lib/index.js");
  if (fs.existsSync(api)) {
    let s = fs.readFileSync(api, "utf8");
    s = s.replaceAll('displayName: "Assistix"', 'displayName: "Beam Agent"');
    s = s.replaceAll('providerName: "Assistix"', 'providerName: "Beam Agent"');
    fs.writeFileSync(api, s);
  }
}
