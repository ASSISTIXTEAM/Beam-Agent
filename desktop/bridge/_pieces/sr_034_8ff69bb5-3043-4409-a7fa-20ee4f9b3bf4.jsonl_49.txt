      Origin: ID_BASE,

    redirect: 'manual',

    const err = new Error(loginJson?.error || `login_http_${loginRes.status}`)

    code = extractOAuthCode(finishLoc)
    if (!code) {

        if (fj.code) code = String(fj.code)

function authDebug(line) {

    fs.appendFileSync(path.join(DATA_DIR, 'bridge.log'), `[auth-password] ${line}\n`)
  } catch {}
}

function extractOAuthError(urlOrText) {
  const raw = String(urlOrText || '')
  if (!raw) return ''
  try {
    const u = new URL(raw, ID_BASE)
    return u.searchParams.get('oauth_error') || u.searchParams.get('error') || ''
  } catch {}
  const m = raw.match(/[?&#]oauth_error=([^&#\s"']+)/i) || raw.match(/[?&#]error=([^&#\s"']+)/i)
  return m ? decodeURIComponent(m[1]) : ''
}

const CHAT_OAUTH_REDIRECT = 'https://chat.assistix.org/auth/callback.php'

async function exchangeCodeWithRedirect(code, redirectUri) {
  const payload = {
    code,
    client_id: OAUTH_CLIENT_ID,
    client_secret: OAUTH_CLIENT_SECRET,
    redirect_uri: redirectUri,
  }
  const res = await fetch(`${ID_BASE}/api/exchange.php`, {
    method: 'POST',
    headers: { 'Content-Type': 'application/json', Accept: 'application/json' },
    body: JSON.stringify(payload),
  })
  const text = await res.text()
  let data
  try { data = JSON.parse(text) } catch { data = { error: text.slice(0, 200) } }
  if (!res.ok) {
    const err = new Error(data?.error || `exchange_http_${res.status}`)
    err.detail = data
    err.status = res.status
    throw err
  }
  if (!isRealIdUser(data?.user)) throw new Error('no_real_user')
  return data
}

async function fetchBootstrap(jar, redirectUri, oauthState) {
  const q = new URLSearchParams({
    redirect_uri: redirectUri,
    client_id: OAUTH_CLIENT_ID,
    state: oauthState,
  })
  const res = await fetch(`${ID_BASE}/api/bootstrap-login.php?${q}`, {
    method: 'GET',
    headers: {
      Accept: 'application/json',
      Origin: ID_BASE,
      Referer: `${ID_BASE}/auth/?${q}`,
      ...(jar.header() ? { Cookie: jar.header() } : {}),
    },
    redirect: 'manual',
  })
  jar.store(res)
  const text = await res.text()
  try { return JSON.parse(text) } catch { return null }
}

async function postOAuthFinish(jar, redirectUri, oauthState) {
  const finishBody = new URLSearchParams({
    redirect_uri: redirectUri,
    client_id: OAUTH_CLIENT_ID,
    state: oauthState,
  })
  const finishRes = await fetch(`${ID_BASE}/api/oauth-finish.php`, {
    method: 'POST',
    headers: {
      'Content-Type': 'application/x-www-form-urlencoded',
      Accept: 'application/json, text/html',
      Origin: ID_BASE,
      Referer: `${ID_BASE}/auth/`,
      ...(jar.header() ? { Cookie: jar.header() } : {}),
    },
    body: finishBody,
    redirect: 'manual',
  })
  jar.store(finishRes)
  const finishLoc = finishRes.headers.get('location') || ''
  const finishText = await finishRes.text()
  let code = extractOAuthCode(finishLoc) || extractOAuthCode(finishText)
  let oauthErr = extractOAuthError(finishLoc) || extractOAuthError(finishText)
  let finishJson = null
  try { finishJson = JSON.parse(finishText) } catch {}
  if (!code && finishJson) {
    if (finishJson.code) code = String(finishJson.code)
    else if (finishJson.redirect) {
      code = extractOAuthCode(finishJson.redirect)
      oauthErr = oauthErr || extractOAuthError(finishJson.redirect)
    }
    if (!oauthErr && finishJson.error) oauthErr = String(finishJson.error)
  }
  return { code, oauthErr, finishLoc, finishJson, status: finishRes.status }
}

/**
 * Primary password login against real ASSISTIX ID.
 * Mirror browser: warm AID session cookie → login-actions → follow redirect →
 * bootstrap → oauth-finish (Beam loopback, then chat redirect fallback) → exchange.
 */
async function loginWithPassword(email, password) {
  const emailRaw = String(email || '').trim()
  const emailNorm = emailRaw.toLowerCase()
  const pass = String(password || '')
  if (!emailNorm.includes('@') || !pass) {
    const err = new Error('email_password_required')
    err.status = 400
    throw err
  }

  const state = loadState()
  if (!state.oauth_nonce) {
    state.oauth_nonce = crypto.randomBytes(16).toString('hex')
    saveState(state)
  }
  const oauthState = oauthStateToken(state.oauth_nonce)
  const beamRedirect = OAUTH_REDIRECT()
  const jar = createCookieJar()

  // 1) Warm anonymous AID session (browser always hits /auth/ first)
  try {
    const warmQ = new URLSearchParams({
      redirect_uri: beamRedirect,
      client_id: OAUTH_CLIENT_ID,
      state: oauthState,
    })
    const warm = await fetch(`${ID_BASE}/auth/?${warmQ}`, {
      method: 'GET',
      headers: { Accept: 'text/html,application/json', Referer: ID_BASE },
      redirect: 'manual',
    })
    jar.store(warm)
    await warm.arrayBuffer().catch(() => {})
    const boot0 = await fetchBootstrap(jar, beamRedirect, oauthState)
    authDebug(`warm cookies=${jar.header() ? 'yes' : 'no'} boot_logged_in=${!!boot0?.logged_in}`)
  } catch (e) {
    authDebug(`warm_fail ${e.message || e}`)
  }

  // 2) Password login against real ID DB
  const fd = new FormData()
  fd.append('action', 'login_password')
  fd.append('email', emailRaw)
  fd.append('password', pass)
  fd.append('redirect_uri', beamRedirect)
  fd.append('client_id', OAUTH_CLIENT_ID)
  fd.append('state', oauthState)

  const loginRes = await fetch(`${ID_BASE}/api/login-actions.php`, {
    method: 'POST',
    body: fd,
    headers: {
      Accept: 'application/json',
      Origin: ID_BASE,
      Referer: `${ID_BASE}/auth/`,
      ...(jar.header() ? { Cookie: jar.header() } : {}),
    },
    redirect: 'manual',
  })
  jar.store(loginRes)
  const loginText = await loginRes.text()
  let loginJson
  try { loginJson = JSON.parse(loginText) } catch { loginJson = null }

  if (!loginJson || loginJson.ok !== true) {
    const err = new Error(loginJson?.error || `login_http_${loginRes.status}`)
    err.status = loginRes.status >= 400 ? loginRes.status : 401
    throw err
  }

  authDebug(
    `login_ok keys=${Object.keys(loginJson).join(',')} redirect=${String(loginJson.redirect || '').slice(0, 120)} hasUser=${isRealIdUser(loginJson.user)} cookies=${jar.header() ? 'yes' : 'no'}`,
  )

  // Prefer any real user/token/code the ID API already returned (no inventing).
  if (loginJson.code) return exchangeCodeWithRedirect(String(loginJson.code), beamRedirect)
  if (isRealIdUser(loginJson.user) && (loginJson.access_token || loginJson.token)) {
    return {
      user: loginJson.user,
      access_token: loginJson.access_token || loginJson.token || null,
      token: loginJson.token || loginJson.access_token || null,
    }
  }

  let candidateUser = isRealIdUser(loginJson.user) ? loginJson.user : null
  let code = extractOAuthCode(loginJson.redirect) || extractOAuthCode(loginRes.headers.get('location') || '')
  let lastOAuthErr = ''

  // 3) Follow login redirect (browser does location.assign(j.redirect))
  if (!code && loginJson.redirect) {
    let followUrl = absIdUrl(loginJson.redirect)
    for (let hop = 0; hop < 5 && followUrl && !code; hop++) {
      const followRes = await fetch(followUrl, {
        method: 'GET',
        headers: {
          Accept: 'text/html, application/json',
          ...(jar.header() ? { Cookie: jar.header() } : {}),
        },
        redirect: 'manual',
      })
      jar.store(followRes)
      const loc = followRes.headers.get('location') || ''
      code = extractOAuthCode(loc)
      lastOAuthErr = extractOAuthError(loc) || lastOAuthErr
      if (code) break
      if (loc) {
        followUrl = absIdUrl(loc)
        continue
      }
      const ft = await followRes.text()
      code = extractOAuthCode(ft)
      lastOAuthErr = extractOAuthError(ft) || lastOAuthErr
      break
    }
  }

  // 4) Bootstrap — confirms session uid + may expose user profile
  const boot = await fetchBootstrap(jar, beamRedirect, oauthState)
  authDebug(`bootstrap logged_in=${!!boot?.logged_in} hasUser=${isRealIdUser(boot?.user)} finish=${!!boot?.oauth_finish}`)
  if (!candidateUser && isRealIdUser(boot?.user)) candidateUser = boot.user

  // 5) oauth-finish: try Beam loopback first, then registered chat redirect
  if (!code) {
    for (const redirectUri of [beamRedirect, CHAT_OAUTH_REDIRECT]) {
      const fin = await postOAuthFinish(jar, redirectUri, oauthState)
      authDebug(`oauth-finish redirect=${redirectUri} status=${fin.status} loc=${String(fin.finishLoc || '').slice(0, 160)} err=${fin.oauthErr || ''}`)
      if (fin.code) {
        try {
          return await exchangeCodeWithRedirect(fin.code, redirectUri)
        } catch (ex) {
          authDebug(`exchange_fail ${ex.message || ex}`)
          lastOAuthErr = String(ex.message || ex)
          continue
        }
      }
      if (fin.oauthErr) lastOAuthErr = fin.oauthErr
      if (fin.finishJson && isRealIdUser(fin.finishJson.user)) {
        return {
          user: fin.finishJson.user,
          access_token: fin.finishJson.access_token || fin.finishJson.token || null,
          token: fin.finishJson.token || fin.finishJson.access_token || null,
        }
      }
    }
  } else {
    return exchangeCodeWithRedirect(code, beamRedirect)
  }

  // 6) Password was accepted by ID — if we have a real profile, enter pending confirm
  if (candidateUser && isRealIdUser(candidateUser)) {
    authDebug(`fallback_user id=${candidateUser.id || candidateUser.id_user_id} email=${candidateUser.email}`)
    return { user: candidateUser, access_token: null, token: null }
  }

  const errMsg = lastOAuthErr
    || (boot && boot.logged_in === false ? 'missing_uid' : '')
    || 'no_oauth_code'
  const err = new Error(errMsg)
  err.status = 502
  throw err
}
