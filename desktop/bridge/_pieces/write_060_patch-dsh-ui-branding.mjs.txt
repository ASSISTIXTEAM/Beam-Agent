/** Word-boundary Beam Agent → Beam after exact pairs (safe: won't touch BEAM_HOME / __BEAM_BOOT__). */
const WORD_PASSES = [
  [/\bDSH\b/g, "Beam"],
  // Lowercase standalone product mention in prose (not package paths with hyphen)
  [/(?<![A-Za-z0-9_/@.-])beam(?![A-Za-z0-9_/-])/g, "beam"],
];

const TARGET_NAME_RE =
  /(client\.js|index\.js|locale\.js|main\.js|\.json|\.yml|\.yaml|\.html|\.md)$/i;

/** Paths that hold user-visible copy (or shell dialogs). */
function isUiCopyPath(file) {
  const n = file.replace(/\\/g, "/").toLowerCase();
  if (n.includes("/node_modules/") && !n.includes("/@assistix/") && !n.includes("/@beam-ai/")) {
    // app lib/main.js etc.
  }
  if (n.endsWith("/lib/main.js") || n.endsWith("/lib/locale.js")) return true;
  if (n.includes("/locale/")) return true;
  if (/\/(beam|beam)-client-ui[^/]*\/lib\/client\.js$/.test(n)) return true;
  if (/\/(beam|beam)-experimental-client-ui[^/]*\/lib\/client\.js$/.test(n)) return true;
  if (/\/(beam|beam)-api-remotes\/lib\/client\.js$/.test(n)) return true;
  if (/\/(beam|beam)-client-ui-agent-preset\/lib\/client\.js$/.test(n)) return true;
  if (/\/(beam|beam)-web-frontend\/dist\/index\.html$/.test(n)) return true;
  if (/\/(beam|beam)-skill-badge\//.test(n)) return true;
  if (/\/(beam|beam)-shell-env\/locale\//.test(n)) return true;
  if (/\/(beam|beam)-experimental-badge-skill-bundle\//.test(n)) return true;
  if (/\/(beam|beam)-llm-pi-ai\/locale\//.test(n)) return true;
  if (/\/(beam|beam)-harness-dev-context\/lib\/index\.js$/.test(n)) return true;
  return false;
}

function rewriteUiCopy(s) {
  let out = s;
  for (const [from, to] of EXACT) {
    if (out.includes(from)) {
      out = out.split(from).join(to);
      report.stringSamples.add(`${from.slice(0, 72)}… → ${to.slice(0, 72)}…`);
    }
  }
  for (const [re, to] of WORD_PASSES) {
    if (re.test(out)) {
      out = out.replace(re, to);
      report.stringSamples.add(`${re} → ${to}`);
    }
  }
  return out;
}

function walk(dir, out = []) {
  if (!fs.existsSync(dir)) return out;
  for (const ent of fs.readdirSync(dir, { withFileTypes: true })) {
    if (ent.name === ".git" || ent.name === "node_modules" && dir.endsWith("node_modules")) {
      // still walk top-level package node_modules
    }
    const p = path.join(dir, ent.name);
    if (ent.isDirectory()) {
      if (ent.name === ".git") continue;
      walk(p, out);
    } else out.push(p);
  }
  return out;
}

function patchFile(file) {
  if (!TARGET_NAME_RE.test(file)) return false;
  if (!isUiCopyPath(file)) return false;
  let s;
  try {
    s = fs.readFileSync(file, "utf8");
  } catch {
    return false;
  }
  if (!/DSH|Beam Agent|\bdsh\b/.test(s)) return false;
  const next = rewriteUiCopy(s);
  if (next === s) return false;
  fs.writeFileSync(file, next);
  report.files.push(file);
  return true;
}

/** Inject / refresh DOM safety net that rewrites residual Beam Agent in live text. */
const DOM_MARK = "beam-dsh-ui-brand-scrub";
const DOM_SNIPPET = `
<script>
/* ${DOM_MARK} */
(function () {
  if (window.__BEAM_DSH_UI_BRAND_SCRUB__) return;
  window.__BEAM_DSH_UI_BRAND_SCRUB__ = true;
  function rewrite(s) {
    if (!s || typeof s !== "string") return s;
    return s
      .replace(/Beam\\s+Harness/gi, "Beam Agent")
      .replace(/\\bDSH\\b/g, "Beam")
      .replace(/(?<![A-Za-z0-9_/@.-])beam(?![A-Za-z0-9_/-])/g, "beam");
  }
  function scrubNode(n) {
    if (!n) return;
    if (n.nodeType === 3) {
      const v = n.nodeValue;
      const nv = rewrite(v);
      if (nv !== v) n.nodeValue = nv;
      return;
    }
    if (n.nodeType !== 1) return;
    const tag = (n.tagName || "").toLowerCase();
    if (tag === "script" || tag === "style" || tag === "noscript") return;
    for (const c of [...n.childNodes]) scrubNode(c);
  }
  function run() {
    try { scrubNode(document.body || document.documentElement); } catch {}
  }
  const mo = new MutationObserver(() => run());
  function start() {
    run();
    try {
      mo.observe(document.documentElement, { childList: true, subtree: true, characterData: true });
    } catch {}
  }
  if (document.readyState === "loading") document.addEventListener("DOMContentLoaded", start);
  else start();
})();
</script>
`;

function patchFrontendHtml(file) {
  if (!fs.existsSync(file)) return false;
  let s = fs.readFileSync(file, "utf8");
  const re = new RegExp(
    `<script>\\s*/\\* ${DOM_MARK} \\*/[\\s\\S]*?</script>\\s*`,
    "g"
  );
  s = s.replace(re, "");
  if (s.includes("</body>")) {
    s = s.replace("</body>", `${DOM_SNIPPET}</body>`);
  } else if (s.includes("</head>")) {
    s = s.replace("</head>", `${DOM_SNIPPET}</head>`);
  } else {
    s += DOM_SNIPPET;
  }
  // Also rewrite any static Beam Agent copy already in the HTML string tables via rewriteUiCopy
  s = rewriteUiCopy(s);
  fs.writeFileSync(file, s);
  if (!report.files.includes(file)) report.files.push(file);
  return true;
}

  let s = fs.readFileSync(file, "utf8");
  if (!s.includes("function scrubBrandLeak")) return false;
  const needle = "t = t.replace(/\\bBeam\\s+(?:Harness|Beam Agent)\\b/gi, 'Beam Agent')";
  const needleAlt = 't = t.replace(/\\bBeam\\s+(?:Harness|Beam Agent)\\b/gi, "Beam Agent")';
  const inject = [
    "t = t.replace(/\\bBeam\\s+(?:Harness|Beam Agent)\\b/gi, 'Beam Agent')",
    "t = t.replace(/\\bDSH\\b/g, 'Beam')",
    "t = t.replace(/The Beam process exited before this turn finished[^.]*(?:\\.\\s*)?/gi, (m) => m) // already Beam",
    "t = t.replace(/\\bBeam\\s+Harness\\s+process\\b/gi, 'Beam process')",
    "t = t.replace(/(?<![A-Za-z0-9_/@.-])beam(?![A-Za-z0-9_/-])/gi, 'beam')",
  ].join("\n  ");
  // Prefer inserting after existing Beam Agent line
  if (s.includes("t = t.replace(/\\bDSH\\b/g, 'Beam')") || s.includes('t = t.replace(/\\bDSH\\b/g, "Beam")')) {
    // already extended
  } else if (s.includes(needle)) {
    s = s.replace(
      needle,
      [
        "t = t.replace(/\\bBeam\\s+(?:Harness|Beam Agent)\\b/gi, 'Beam Agent')",
        "t = t.replace(/\\bDSH\\b/g, 'Beam')",
        "t = t.replace(/\\bBeam\\s+Harness\\s+process\\b/gi, 'Beam process')",
        "t = t.replace(/(?<![A-Za-z0-9_/@.-])beam(?![A-Za-z0-9_/-])/gi, 'beam')",
      ].join("\n  ")
    );
  } else if (s.includes("Beam\\s+(?:Harness|Beam Agent)")) {
    s = s.replace(
      /t = t\.replace\(\/\\bBeam\\s\+\(\?:Harness\|Beam Agent\)\\b\/gi,\s*['"]Beam Agent['"]\)/,
      [
        "t = t.replace(/\\bBeam\\s+(?:Harness|Beam Agent)\\b/gi, 'Beam Agent')",
        "t = t.replace(/\\bDSH\\b/g, 'Beam')",
        "t = t.replace(/\\bBeam\\s+Harness\\s+process\\b/gi, 'Beam process')",
        "t = t.replace(/(?<![A-Za-z0-9_/@.-])beam(?![A-Za-z0-9_/-])/gi, 'beam')",
      ].join("\n  ")
    );
  } else {
    // Insert before return t in scrubBrandLeak
    s = s.replace(
      /(function scrubBrandLeak\(text\) \{[\s\S]*?)(\n\s*return t\n\})/,
      `$1\n  t = t.replace(/\\bBeam\\s+(?:Harness|Beam Agent)\\b/gi, 'Beam Agent')\n  t = t.replace(/\\bDSH\\b/g, 'Beam')\n  t = t.replace(/(?<![A-Za-z0-9_/@.-])beam(?![A-Za-z0-9_/-])/gi, 'beam')\n$2`
    );
  }
  // Drop the noop line if we accidentally added a weird comment form
  s = s.replace(
    /\n\s*t = t\.replace\(\/The Beam process exited before this turn finished\[\^\.\]\*\(\?:\\\.\\s\*\)\?\/gi, \(m\) => m\) \/\/ already Beam/g,
    ""
  );
  fs.writeFileSync(file, s);
  if (!report.files.includes(file)) report.files.push(file);
  return true;
}
